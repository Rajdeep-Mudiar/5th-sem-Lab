    <main>
      <h1>Memory Matching Game</h1>
      <p>Find all matching pairs.</p>
      <p>
        Attempts: <strong id="attempts">0</strong> | Matches:
        <strong id="matches">0</strong>/8
      </p>
      <div id="board" class="board"></div>
      <p id="message" role="status"></p>
      <button onclick="startGame()">Restart Game</button>
    </main>

    <script>
      const symbols = ["🍎", "🌟", "🐱", "🌈", "🎈", "🍀", "🚀", "🎵"];
      let cards = [];
      let firstCard = null;
      let secondCard = null;
      let locked = false;
      let attempts = 0;
      let matches = 0;
      let gameId = 0;
      function startGame() {
        gameId++; // Ignore delayed flips from an older game.
        cards = [...symbols, ...symbols];
        // Shuffle the cards with Fisher-Yates.
        for (let i = cards.length - 1; i > 0; i--) {
          const j = Math.floor(Math.random() * (i + 1));
          [cards[i], cards[j]] = [cards[j], cards[i]];
        }
        firstCard = null;
        secondCard = null;
        locked = false;
        attempts = 0;
        matches = 0;
        document.getElementById("attempts").textContent = "0";
        document.getElementById("matches").textContent = "0";
        document.getElementById("message").textContent = "";
        const board = document.getElementById("board");
        board.innerHTML = "";
        cards.forEach(function (symbol) {
          const card = document.createElement("button");
          card.className = "memory-card";
          card.textContent = "?";
          card.setAttribute("aria-label", "Hidden card");
          card.onclick = function () {
            flipCard(card, symbol);
          };
          board.appendChild(card);
        });
      }
      function flipCard(card, symbol) {
        if (locked || card === firstCard || card.classList.contains("matched"))
          return;
        card.textContent = symbol;
        card.setAttribute("aria-label", symbol);
        if (firstCard === null) {
          firstCard = card;
          return;
        }
        secondCard = card;
        attempts++;
        document.getElementById("attempts").textContent = attempts;
        if (firstCard.textContent === secondCard.textContent) {
          firstCard.classList.add("matched");
          secondCard.classList.add("matched");
          firstCard.disabled = true;
          secondCard.disabled = true;
          matches++;
          document.getElementById("matches").textContent = matches;
          firstCard = null;
          secondCard = null;
          if (matches === symbols.length)
            document.getElementById("message").textContent =
              "Congratulations! You found all pairs.";
        } else {
          locked = true;
          const oldGame = gameId;
          setTimeout(function () {
            if (oldGame !== gameId) return;
            firstCard.textContent = "?";
            secondCard.textContent = "?";
            firstCard.setAttribute("aria-label", "Hidden card");
            secondCard.setAttribute("aria-label", "Hidden card");
            firstCard = null;
            secondCard = null;
            locked = false;
          }, 900);
        }
      }
      startGame();
    </script>
